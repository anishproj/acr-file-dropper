void selectionSort(int a [10] ,int size) {

	
	int temp;
	for (int i = 0; i < size - 1; i++) {
		int min = i;
		for (int j = i + 1; j < size; j++) {
			if (a[j] < a[min]) {
				min = j;
			}
		}
		if (min != i) {
			temp = a[i];
			a[i] = a[min];
			a[min] = temp;
		}
		

	}

	printf("ouput: \n");

	for (int i = 0; i < size; i++) {
		printf("%d\t", a[i]);
	}



}

void linearSearch(int a [10], int key,int size) {

	for (int i = 0; i < size; i++) if (a[i] == key) printf("found key!");

	else	printf("not found!!\n");


}


void binarySearch(int a[10],int key,int size) {
	int low = 0, high = size - 1;
	while (low != high) {
		int mid = low + (high - low) / 2;

		if (a[mid] == key) {
			printf("found value %d! \n", key);
			break;
		}
		else if (a[mid] > key) {
			high = mid - 1;
		}
		else  low = mid + 1;
		

	}
	printf("not found!");

}

//----------------------------- recursive
void binarySearchRecursive(int a[10], int low, int high , int key) {
	if (!(low >= high)) {
	
			int mid = low + (high - low) / 2;

			if (a[mid] == key) {
				printf("found value %d! \n", key);
				return;
			}
			else if (a[mid] > key) {
				binarySearchRecursive(a, low, mid - 1, key);
			}
			else  binarySearchRecursive(a, mid+1, high, key);


		

			printf("not found \n");
	}

	

}

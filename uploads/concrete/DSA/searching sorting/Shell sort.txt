void shellSort(int a[10] , int size) {

	int g = size / 2;
	int temp, swapped, i;

	do {

		do {
			swapped = 0;
			for (i = 0; i < size - g; i++) {
				if (a[i] > a[i + g]) {
					temp = a[i];
					a[i] = a[i+g];
					a[i + g] = temp;
					swapped = 1;

				}
			}
		} while (swapped == 1);

			g /= 2;
	} while (g >= 1);
	
	for (int i = 0; i < size; i++) {
		printf("%d\t", a[i]);
	}

}

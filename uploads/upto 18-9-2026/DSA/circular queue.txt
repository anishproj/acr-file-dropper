#include <stdio.h>
#include <string.h>
#include <stdbool.h>

struct job
{
	int job;
	char title[20];
};

bool isFull(int front, int rear, int size)
{
	if((rear + 1) % size == front)
		return 1;
	return 0;
}

bool isEmpty(int front, int rear, int size)
{	
	if(front == rear)
		return 1;
	return 0;
}

void enqueue(struct job queue[5], int *rear, int size)
{
	
	*rear = (*rear + 1) % size;
	printf("Enter job ID and document title :- ");
	scanf("%d", &queue[*rear].job);
	scanf("%s", queue[*rear].title);

}

void dequeue(struct job queue[5], int *front, int size)
{
	
	*front = (*front + 1) % size;
	printf("%d: %s is removed\n", queue[*front].job, queue[*front].title);
}

void display(struct job queue[5], int front, int rear, int size)
{
	for(int i = (front + 1) % size; i != rear ; i = (i+1) % size)
		printf("%d: %s, ", queue[i].job, queue[i].title);
	
	printf("%d: %s, ", queue[rear].job, queue[rear].title);
	printf("\n");
}

int main()
{
	struct job queue[5];
	int num, rear = 0, front = 0, size = 5;
	while(1)
	{
		int ptr;
		printf("1: enqueue\n2: dequeue\n3: display\n4: break\nEnter :- ");
		scanf("%d", &ptr);
		switch(ptr)
		{
			case 1:
				if(isFull(front, rear, size))
					printf("queue is full\n");
				else
					enqueue(queue, &rear, size);
				break;
				
			case 2:
				if(isEmpty(front, rear, size))
					printf("queue is empty\n");
				else
					dequeue(queue, &front, size);
				break;
				
			case 3:
				if(isEmpty(front, rear, size))
					printf("queue is empty\n");
				else
					display(queue, front, rear, size);
				break;
				
			case 4:
				return 0;
		}
	
	}


	return 0;
}
